STEP 01 · WHY FILES?§7.1

왜 파일이 필요한가?

메모리에 저장된 변수는 일반적으로 프로그램이 끝나면 사라집니다. 반면 파일에 기록한 정보는 다음 실행에서 다시 읽을 수 있습니다. 이것이 File I/O가 필요한 이유입니다.

참고 C 코드에서 fopen(), fprintf(), fclose()의 실패 처리 위치를 찾아보세요.

Core idea

int value 같은 변수는 실행 중인 프로그램의 메모리 상태를 나타냅니다. 저장된 파일은 프로그램 외부의 데이터를 표현하므로 다음 실행에서도 사용할 수 있습니다.

참고 코드 펼치기 · 프로그램 종료 후 파일 저장 원리
BEGINNER STYLE · OPEN → CHECK → WRITE → CLOSE · complete C example
#include <stdio.h>

int main(void) {
    double score = 92.5; /* value in process memory */
    FILE *fp;

    fp = fopen("scores.txt", "w");
    if (fp == NULL) {
        perror("scores.txt");
        return 1;
    }

    if (fprintf(fp, "101 Mina %.1f\n", score) < 0) {
        fprintf(stderr, "File write error\n");
        fclose(fp);
        return 1;
    }

    if (fclose(fp) == EOF) {
        perror("fclose");
        return 1;
    }

    return 0;
}

Complete C program
#include <stdio.h>
int main(void) {
    FILE *fp;
    char word[20], line[80];
    int id;
    double score;
    fp = fopen("input.txt", "r");
    if (fp == NULL) return 1;
    if (fscanf(fp, "%19s", word) == 1)
        printf("A: %s\n", word);
    fclose(fp);

    fp = fopen("input.txt", "r");
    if (fp == NULL) return 1;
    if (fgets(line, sizeof line, fp) != NULL)
        printf("B: %s", line);
    fclose(fp);

    fp = fopen("student.txt", "r");
    if (fp == NULL) return 1;
    printf("Conversions: %d\n",
           fscanf(fp, "%d %lf", &id, &score));
    fclose(fp);
    return 0;
}

#include <stdio.h>
#include <stdlib.h>

int main(void) {
    const double threshold = 3.0;
    double time_s, value, sum = 0.0;
    int count = 0;
    int fields;
    int result = EXIT_SUCCESS;

    FILE *fp = fopen("sensor_log.txt", "r");
    if (fp == NULL) {
        perror("sensor_log.txt");
        return EXIT_FAILURE;
    }

    while ((fields = fscanf(fp, "%lf %lf", &time_s, &value)) == 2) {
        sum += value;
        count++;
        if (value > threshold) {
            printf("ALERT %.1f: %.2f\n", time_s, value);
        }
    }

    if (ferror(fp)) {
        fprintf(stderr, "Sensor file read error\n");
        result = EXIT_FAILURE;
    } else if (fields != EOF) {
        fprintf(stderr, "Invalid sensor record: %d field(s) read\n", fields);
        result = EXIT_FAILURE;
    }

    if (fclose(fp) == EOF) {
        perror("fclose");
        result = EXIT_FAILURE;
    }
    if (result != EXIT_SUCCESS) {
        return result;
    }
    if (count > 0) {
        printf("Count = %d\n", count);
        printf("Average = %.3f\n", sum / count);
    } else {
        puts("No sensor records");
    }
    return EXIT_SUCCESS;
}


코드 스타일: 실무에서 흔히 볼 수 있는 FILE *fp = fopen(...); 초기화 → if (fp == NULL) 검사 → 입력 반환값 검사 → fclose() 검사 형식입니다. fields != EOF는 파일 끝이 아닌 불완전·부정확한 센서 레코드를 탐지합니다. 입력 도중 실제 I/O 오류는 ferror(fp)로 별도 검사합니다.
