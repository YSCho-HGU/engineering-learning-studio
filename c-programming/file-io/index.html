7.1 · WHY IT MATTERS

프로그램을 다시 실행하면 데이터는 어디에 있을까요?

메모리에 저장된 변수는 일반적으로 프로그램이 끝나면 사라집니다. 반면 파일에 기록한 정보는 다음 실행에서 다시 읽을 수 있습니다. 이것이 File I/O가 필요한 이유입니다.

Core idea

int value 같은 변수는 실행 중인 프로그램의 메모리 상태를 나타냅니다. 저장된 파일은 프로그램 외부의 데이터를 표현하므로 다음 실행에서도 사용할 수 있습니다.

7.2 · STREAM MODEL

파일 I/O는 새로운 개념이 아니라 Stream의 확장입니다

이미 사용해 온 scanf()와 printf()도 Stream을 통해 데이터를 주고받습니다. 무엇을 읽고 어디에 쓸지만 달라질 뿐입니다.

Core idea

stdin은 표준 입력, stdout은 정상 출력, stderr는 오류·진단 출력입니다. FILE *fp는 열린 Stream을 가리키는 포인터이며 파일 이름 자체를 담고 있는 문자열 변수가 아닙니다.

7.5 · STRUCTURE RECORDS
구조체는 메모리에, 파일에는 문자로 저장됩니다
Structure를 File I/O와 연결하려면 기록할 텍스트 형식을 결정해야 합니다. 저장할 때와 읽을 때 같은 멤버 순서를 사용하세요.


COMPLETE C EXAMPLE · write, close, reopen, read
#include <stdio.h>

struct Student {
    int id;
    char name[40];
    double score;
};

int main(void) {
    struct Student s = {101, "Mina", 92.5};
    FILE *fp;

    fp = fopen("students.txt", "w");
    if (fp == NULL) return 1;
    if (fprintf(fp, "%d %s %.1f\n",
                s.id, s.name, s.score) < 0) {
        fclose(fp);
        return 1;
    }
    if (fclose(fp) == EOF) return 1;

    fp = fopen("students.txt", "r");
    if (fp == NULL) return 1;
    if (fscanf(fp, "%d %39s %lf",
               &s.id, s.name, &s.score) != 3) {
        fprintf(stderr, "Invalid student record\n");
        fclose(fp);
        return 1;
    }
    printf("%d %s %.1f\n", s.id, s.name, s.score);
    if (fclose(fp) == EOF) return 1;
    return 0;
}

7.5.3 · ENGINEERING APPLICATION
센서 데이터로 File I/O의 활용을 확인해 봅시다
온도·전압·환경 센서에서는 데이터를 시간 순서대로 저장합니다. 여기서는 Sensor Log를 읽어 평균을 구하고 임계값을 넘는 측정치를 찾아봅니다.


COMPLETE C EXAMPLE · sensor_log.txt contains time and value pairs
#include <stdio.h>

int main(void) {
    FILE *fp;
    double t, value, sum = 0.0;
    double threshold = 3.0;
    int count = 0;

    fp = fopen("sensor_log.txt", "r");
    if (fp == NULL) return 1;

    while (fscanf(fp, "%lf %lf", &t, &value) == 2) {
        sum += value;
        count++;
        if (value > threshold)
            printf("ALERT %.1f: %.2f\n", t, value);
    }
    if (ferror(fp)) {
        fprintf(stderr, "Sensor file read error\n");
        fclose(fp);
        return 1;
    }
    if (!feof(fp)) {
        fprintf(stderr, "Invalid sensor data\n");
        fclose(fp);
        return 1;
    }
    if (fclose(fp) == EOF) return 1;
    if (count > 0)
        printf("Average = %.3f\n", sum / count);
    return 0;
}
