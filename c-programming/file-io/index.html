7.1 · WHY IT MATTERS

프로그램을 다시 실행하면 데이터는 어디에 있을까요?

#include <stdio.h>

int main(void) {
    double score = 92.5; /* value in process memory */
    FILE *fp;

    fp = fopen("scores.txt", "w");
    if (fp == NULL) {
        perror("scores.txt");
        return 1;
    }

    if (fprintf(fp, "101 Mina %.1f\n", score) < 0) {
        fprintf(stderr, "File write error\n");
        fclose(fp);
        return 1;
    }

    if (fclose(fp) == EOF) {
        perror("fclose");
        return 1;
    }

    return 0;
}

참고 코드 펼치기 · Structure 저장·복원 전체 C 예제
MIDDLE STYLE · INITIALIZE + CHECK · write, close, reopen, read
#include <stdio.h>

struct Student {
    int id;
    char name[40];
    double score;
};

int main(void) {
    struct Student s = {101, "Mina", 92.5};
    FILE *fp = fopen("students.txt", "w");
    if (fp == NULL) {
        perror("students.txt");
        return 1;
    }
    if (fprintf(fp, "%d %s %.1f\n",
                s.id, s.name, s.score) < 0) {
        fclose(fp);
        return 1;
    }
    if (fclose(fp) == EOF) {
        perror("fclose");
        return 1;
    }

    fp = fopen("students.txt", "r");
    if (fp == NULL) {
        perror("students.txt");
        return 1;
    }
    if (fscanf(fp, "%d %39s %lf",
               &s.id, s.name, &s.score) != 3) {
        fprintf(stderr, "Invalid student record\n");
        fclose(fp);
        return 1;
    }
    printf("%d %s %.1f\n", s.id, s.name, s.score);
    if (fclose(fp) == EOF) {
        perror("fclose");
        return 1;
    }
    return 0;
}

PRACTICAL C11 EXAMPLE · GCC verified · 아래 threshold와 코드 연동
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    const double threshold = 3.0;
    double time_s, value, sum = 0.0;
    int count = 0;
    int fields;
    int result = EXIT_SUCCESS;

    FILE *fp = fopen("sensor_log.txt", "r");
    if (fp == NULL) {
        perror("sensor_log.txt");
        return EXIT_FAILURE;
    }

    while ((fields = fscanf(fp, "%lf %lf", &time_s, &value)) == 2) {
        sum += value;
        count++;
        if (value > threshold) {
            printf("ALERT %.1f: %.2f\n", time_s, value);
        }
    }

    if (ferror(fp)) {
        fprintf(stderr, "Sensor file read error\n");
        result = EXIT_FAILURE;
    } else if (fields != EOF) {
        fprintf(stderr, "Invalid sensor record: %d field(s) read\n", fields);
        result = EXIT_FAILURE;
    }

    if (fclose(fp) == EOF) {
        perror("fclose");
        result = EXIT_FAILURE;
    }
    if (result != EXIT_SUCCESS) {
        return result;
    }
    if (count > 0) {
        printf("Count = %d\n", count);
        printf("Average = %.3f\n", sum / count);
    } else {
        puts("No sensor records");
    }
    return EXIT_SUCCESS;
}


코드 스타일: 실무에서 흔히 볼 수 있는 FILE *fp = fopen(...); 초기화 → if (fp == NULL) 검사 → 입력 반환값 검사 → fclose() 검사 형식입니다. fields != EOF는 파일 끝이 아닌 불완전·부정확한 센서 레코드를 탐지합니다. 입력 도중 실제 I/O 오류는 ferror(fp)로 별도 검사합니다.
